```ts
// src/manifest.ts
import { z } from 'zod';
export interface JobSpec {
  name: string;                         // unique within module
  intervalSec: number;                  // >= 5
  mode: 'always' | 'whenViewed';
  per: 'instance' | 'module';
  timeoutSec?: number;                  // default 30
}
export type OutboundRule =
  | { kind: 'host'; host: string }      // exact hostname, https only
  | { kind: 'publicUrl' }               // user-supplied URL, must resolve public
  | { kind: 'lanDevice' };              // admin-configured private address
export interface RootMount { prefix: string; auth: 'session' | 'appPassword' }
export interface ModuleManifest<
  S extends z.ZodType = z.ZodType,      // per-instance settings
  C extends z.ZodType = z.ZodType       // module-wide admin config
> {
  id: ModuleId;
  name: string;                         // English fallback; shown via t(`module.${id}.name`, name)
  description: string;
  version: string;                      // semver
  icon: string;                         // icon key, see theme.ts ICON_KEYS or module-provided svg key
  scope: 'user' | 'shared';
  multiInstance: boolean;
  tile: { default: TileSize; min: TileSize; max: TileSize; staleAfterSec?: number };
  settings: S;
  config: C;
  levels: ReadonlyArray<'view' | 'edit'>;   // which grants make sense
  jobs: ReadonlyArray<JobSpec>;
  publishes: ReadonlyArray<EventName>;
  subscribes: ReadonlyArray<EventName>;
  outbound: ReadonlyArray<OutboundRule>;
  rootMounts?: ReadonlyArray<RootMount>;    // core allowlists these per module id
}
export function validateManifest(m: ModuleManifest): string[];   // returns problems; [] = valid
```

`validateManifest` checks: id matches `ID_RE`; version is semver; `min ≤ default ≤ max` and `w ≤ GRID_COLUMNS`; job names unique; `intervalSec ≥ 5`; every `host` rule has no scheme/path; `publishes`/`subscribes` are keys of `events`; settings/config are `z.object`.

```ts
export const UserDto        = z.object({ id: z.string(), name: z.string(), handle: z.string(), role: z.enum(['admin','member']) });
export const MeDto          = UserDto.extend({ grants: z.record(z.string(), z.enum(['none','view','edit'])), csrf: z.string(),
                                prefs: z.object({ theme: z.string(), modePolicy: z.enum(['day','night','system','sun']), locale: z.string(), timeZone: z.string() }) });
export const LayoutDto      = z.object({ x: z.number().int(), y: z.number().int(), w: z.number().int(), h: z.number().int() });
export const InstanceDto    = z.object({ id: z.string(), moduleId: z.string(), pageId: z.string(), settings: z.unknown(),
                                layout: LayoutDto, mobileOrder: z.number().int() });
export const PageDto        = z.object({ id: z.string(), name: z.string(), icon: z.string().nullable(), order: z.number().int(),
                                ownerId: z.string(), shared: z.boolean(), background: z.string().nullable(), instances: z.array(InstanceDto) });
export const SnapshotDto    = z.object({ instanceId: z.string(), data: z.unknown(), updatedAt: z.string().nullable(), stale: z.boolean() });
export const NotificationDto= z.object({ id: z.string(), moduleId: z.string(), level: z.enum(['info','warn','critical']), title: z.string(),
                                body: z.string().nullable(), url: z.string().nullable(), createdAt: z.string(), readAt: z.string().nullable() });
export const StreamEvent    = z.discriminatedUnion('type', [
  z.object({ type: z.literal('snapshot'), snapshot: SnapshotDto }),
  z.object({ type: z.literal('notification'), notification: NotificationDto }),
  z.object({ type: z.literal('layout'), pageId: z.string() }),
  z.object({ type: z.literal('revoked'), moduleId: z.string() }),
  z.object({ type: z.literal('ping'), at: z.string() }),
]);
export const ErrorDto       = z.object({ error: z.object({ code: z.string(), message: z.string(), details: z.unknown().optional() }) });
```
